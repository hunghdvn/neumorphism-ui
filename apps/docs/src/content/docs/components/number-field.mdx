## Label, help text and error text

`error-text` replaces the browser's validation message once the field is invalid after user
interaction; it does not make the field invalid by itself. The stepper buttons are named by
`increment-label` and `decrement-label`.

<Demo
  code={`<neu-number-field label="Age" placeholder="18" help-text="In years"></neu-number-field>
<neu-number-field label="Quantity" required error-text="Enter a quantity"></neu-number-field>`}
/>

## React

<Code
  lang="tsx"
  code={`import { NeuNumberField } from '@neumorphism-ui/react';

export function QuantityInput({ onChange }: { onChange: (value: number | null) => void }) {
  return (
    <NeuNumberField
      placeholder="Enter quantity"
      onChange={(e) => {
        const { value } = e.target as HTMLElement & { value: string };
        onChange(value === '' ? null : Number(value));
      }}
    />
  );
}`}
/>
