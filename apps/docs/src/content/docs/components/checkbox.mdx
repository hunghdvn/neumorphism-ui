## React

<Code
  lang="tsx"
  code={`import { NeuCheckbox } from '@neumorphism-ui/react';

export function TermsCheckbox({ onChange }: { onChange: (checked: boolean) => void }) {
  return (
    <NeuCheckbox
      onInput={(e) => {
        const target = e.target as HTMLElement & { checked: boolean };
        onChange(target.checked);
      }}
    >
      Accept terms and conditions
    </NeuCheckbox>
  );
}`}
/>
