Set the `pill` attribute to round the badge into a capsule shape.

<Demo
  code={`<neu-badge variant="neutral">Neutral</neu-badge>

/>

## Content

The badge wraps its content in the default slot.

<Demo
  code={`<neu-badge variant="success">12</neu-badge>

/>

## React

<Code
  lang="tsx"
  code={`import { NeuBadge } from '@neumorphism-ui/react';

export function StatusBadge({ status }: { status: 'success' | 'warning' | 'danger' }) {
  const variant = status === 'success' ? 'success' : status === 'warning' ? 'warning' : 'danger';
  return (
    <NeuBadge variant={variant} pill={status === 'danger'}>
      {status.toUpperCase()}
    </NeuBadge>
  );
}`}
/>
